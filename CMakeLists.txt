cmake_minimum_required(VERSION 3.20)
project(LocalAgent LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
    message(FATAL_ERROR "Windows is required")
endif()
get_filename_component(COMPILER_BIN_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_path(SQLITE_INCLUDE_DIR sqlite3.h HINTS "${COMPILER_BIN_DIR}/../include" REQUIRED)
find_library(SQLITE_LIBRARY NAMES libsqlite3.a sqlite3 HINTS "${COMPILER_BIN_DIR}/../lib" REQUIRED)
add_library(rag_core STATIC src/rag_chat.cpp src/ollama_client.cpp src/document_import.cpp src/generation_policy.cpp src/rag_context.cpp)
include(cmake/CppHttplib.cmake)
add_library(local_api STATIC src/chat_api.cpp src/web_chat.cpp)
target_include_directories(local_api PUBLIC include third_party)
target_include_directories(local_api SYSTEM PUBLIC "${LOCAL_AGENT_HTTPLIB_DIR}")
target_compile_definitions(local_api PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(local_api PUBLIC ws2_32)
add_executable(local_agent_server src/server_main.cpp)
target_link_libraries(local_agent_server PRIVATE local_api rag_core)
if(MSVC)
    target_compile_options(local_api PRIVATE /W4 /utf-8)
    target_compile_options(local_agent_server PRIVATE /W4 /utf-8)
else()
    target_compile_options(local_api PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(local_agent_server PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(local_agent_server PRIVATE -static -static-libgcc -static-libstdc++)
endif()
target_include_directories(rag_core PUBLIC include third_party ${SQLITE_INCLUDE_DIR})
target_compile_definitions(rag_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(rag_core PUBLIC winhttp ${SQLITE_LIBRARY})
if(MSVC)
    target_compile_options(rag_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(rag_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(rag_eval EXCLUDE_FROM_ALL tools/rag_eval.cpp)
target_include_directories(rag_eval PRIVATE include third_party ${SQLITE_INCLUDE_DIR})
target_compile_definitions(rag_eval PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(rag_eval PRIVATE rag_core)
if(MSVC)
    target_compile_options(rag_eval PRIVATE /W4 /utf-8)
else()
    target_compile_options(rag_eval PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(rag_eval PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_executable(local_agent WIN32 src/main.cpp src/chat_window.cpp)
target_compile_features(local_agent PRIVATE cxx_std_17)
target_include_directories(local_agent PRIVATE include third_party ${SQLITE_INCLUDE_DIR})
target_compile_definitions(local_agent PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(local_agent PRIVATE rag_core comctl32 comdlg32 dwmapi uxtheme)
find_program(LOCALAGENT_PYTHON NAMES python3 python HINTS "${COMPILER_BIN_DIR}")
if(LOCALAGENT_PYTHON)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/runtime_config")
    file(WRITE "${CMAKE_BINARY_DIR}/runtime_config/python_path.txt" "${LOCALAGENT_PYTHON}\n")
endif()
add_custom_target(rag_assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/config" "$<TARGET_FILE_DIR:local_agent>/config"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:local_agent>/data"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/data/document_index.db" "$<TARGET_FILE_DIR:local_agent>/data/document_index.db"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/scripts" "$<TARGET_FILE_DIR:local_agent>/scripts"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/third_party/python" "$<TARGET_FILE_DIR:local_agent>/third_party/python")
if(LOCALAGENT_PYTHON)
    add_custom_command(TARGET rag_assets POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_BINARY_DIR}/runtime_config/python_path.txt" "$<TARGET_FILE_DIR:local_agent>/config/python_path.txt")
endif()
add_dependencies(local_agent rag_assets)
include(CTest)

    if(LOCALAGENT_PYTHON)
        add_test(NAME document_extraction COMMAND "${LOCALAGENT_PYTHON}"
            "${CMAKE_SOURCE_DIR}/tests/document_extraction_tests.py" "${CMAKE_SOURCE_DIR}/scripts/extract_document.py")
