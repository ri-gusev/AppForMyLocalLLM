cmake_minimum_required(VERSION 3.20)
project(LocalAgent LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Windows is required")
endif()
add_executable(local_agent src/main.cpp src/ollama_client.cpp)
target_compile_features(local_agent PRIVATE cxx_std_17)
target_include_directories(local_agent PRIVATE include third_party)
target_compile_definitions(local_agent PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(local_agent PRIVATE winhttp)
if(MSVC)
    target_compile_options(local_agent PRIVATE /W4 /utf-8)
else()
    target_compile_options(local_agent PRIVATE -Wall -Wextra -Wpedantic)
endif()
